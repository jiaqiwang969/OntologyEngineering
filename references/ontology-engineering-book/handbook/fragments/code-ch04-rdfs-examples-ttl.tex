\codeline{\textcolor{cmtgray}{\#\ RDFS\ (RDF\ Schema)\ 示例\ -\ 制造领域}}
\codeline{\textcolor{cmtgray}{\#\ RDFS\ Examples\ -\ Manufacturing\ Domain}}
\codeline{\textcolor{cmtgray}{\#}}
\codeline{\textcolor{cmtgray}{\#\ RDFS\ 在\ RDF\ 基础上增加了类定义、属性定义和基本的推理能力}}
\codeline{\textcolor{cmtgray}{\#\ 是\ OWL\ 的基础}}
\codeblank{}
\codeline{@prefix\ rdf:\ \ <http://www.w3.org/1999/02/22-rdf-syntax-ns\#>\ .}
\codeline{@prefix\ rdfs:\ <http://www.w3.org/2000/01/rdf-schema\#>\ .}
\codeline{@prefix\ xsd:\ \ <http://www.w3.org/2001/XMLSchema\#>\ .}
\codeline{@prefix\ mfg:\ \ <http://example.org/manufacturing\#>\ .}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{\textcolor{cmtgray}{\#\ 类定义\ (rdfs:Class)}}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeblank{}
\codeline{mfg:Resource\ a\ rdfs:Class\ ;}
\codeline{\hspace*{4\ttcw}rdfs:label\ "资源"@zh\ ;}
\codeline{\hspace*{4\ttcw}rdfs:comment\ "制造系统中可分配的所有资源"@zh\ .}
\codeblank{}
\codeline{mfg:Equipment\ a\ rdfs:Class\ ;}
\codeline{\hspace*{4\ttcw}rdfs:subClassOf\ mfg:Resource\ ;}
\codeline{\hspace*{4\ttcw}rdfs:label\ "设备"@zh\ ;}
\codeline{\hspace*{4\ttcw}rdfs:comment\ "物理设备"@zh\ .}
\codeblank{}
\codeline{mfg:Tool\ a\ rdfs:Class\ ;}
\codeline{\hspace*{4\ttcw}rdfs:subClassOf\ mfg:Resource\ ;}
\codeline{\hspace*{4\ttcw}rdfs:label\ "刀具"@zh\ .}
\codeblank{}
\codeline{mfg:Material\ a\ rdfs:Class\ ;}
\codeline{\hspace*{4\ttcw}rdfs:label\ "材料"@zh\ .}
\codeblank{}
\codeline{mfg:Process\ a\ rdfs:Class\ ;}
\codeline{\hspace*{4\ttcw}rdfs:label\ "工艺"@zh\ .}
\codeblank{}
\codeline{mfg:WorkCell\ a\ rdfs:Class\ ;}
\codeline{\hspace*{4\ttcw}rdfs:label\ "工作单元"@zh\ .}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{\textcolor{cmtgray}{\#\ 属性定义\ (rdf:Property)}}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ rdfs:domain\ 指定属性的主语类型}}
\codeline{\textcolor{cmtgray}{\#\ rdfs:range\ \ 指定属性的宾语类型}}
\codeblank{}
\codeline{mfg:canProcess\ a\ rdf:Property\ ;}
\codeline{\hspace*{4\ttcw}rdfs:label\ "能加工"@zh\ ;}
\codeline{\hspace*{4\ttcw}rdfs:domain\ mfg:Equipment\ ;}
\codeline{\hspace*{4\ttcw}rdfs:range\ mfg:Material\ .}
\codeblank{}
\codeline{mfg:locatedIn\ a\ rdf:Property\ ;}
\codeline{\hspace*{4\ttcw}rdfs:label\ "位于"@zh\ ;}
\codeline{\hspace*{4\ttcw}rdfs:domain\ mfg:Equipment\ ;}
\codeline{\hspace*{4\ttcw}rdfs:range\ mfg:WorkCell\ .}
\codeblank{}
\codeline{mfg:uses\ a\ rdf:Property\ ;}
\codeline{\hspace*{4\ttcw}rdfs:label\ "使用"@zh\ ;}
\codeline{\hspace*{4\ttcw}rdfs:domain\ mfg:Process\ ;}
\codeline{\hspace*{4\ttcw}rdfs:range\ mfg:Tool\ .}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{\textcolor{cmtgray}{\#\ RDFS推理示例}}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{\textcolor{cmtgray}{\#\ 基于以下事实：}}
\codeline{\textcolor{cmtgray}{\#\ \ \ mfg:Lathe\ rdfs:subClassOf\ mfg:Equipment}}
\codeline{\textcolor{cmtgray}{\#\ \ \ mfg:Lathe\_001\ rdf:type\ mfg:Lathe}}
\codeline{\textcolor{cmtgray}{\#\ RDFS推理器可以推出：}}
\codeline{\textcolor{cmtgray}{\#\ \ \ mfg:Lathe\_001\ rdf:type\ mfg:Equipment\ \ (类继承推理)}}
\codeblank{}
\codeline{mfg:Lathe\ a\ rdfs:Class\ ;}
\codeline{\hspace*{4\ttcw}rdfs:subClassOf\ mfg:Equipment\ ;}
\codeline{\hspace*{4\ttcw}rdfs:label\ "车床"@zh\ .}
\codeblank{}
\codeline{mfg:Lathe\_001\ a\ mfg:Lathe\ ;}
\codeline{\hspace*{4\ttcw}rdfs:label\ "1号车床"@zh\ ;}
\codeline{\hspace*{4\ttcw}mfg:locatedIn\ mfg:WorkCell\_A\ .}
\codeline{\textcolor{cmtgray}{\#\ \(\rightarrow\)\ 通过\ rdfs:subClassOf\ 推理，可得\ mfg:Lathe\_001\ a\ mfg:Equipment}}
\codeblank{}
\codeline{mfg:WorkCell\_A\ a\ mfg:WorkCell\ ;}
\codeline{\hspace*{4\ttcw}rdfs:label\ "A区工作单元"@zh\ .}
