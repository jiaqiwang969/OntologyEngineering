\codeblank{}
\codeline{PREFIX\ rdf:\ \ <http://www.w3.org/1999/02/22-rdf-syntax-ns\#>}

\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{\textcolor{cmtgray}{\#\ CQ1:\ 哪些设备可以加工铝合金？}}
\codeline{\textcolor{cmtgray}{\#\ Which\ equipment\ can\ process\ aluminum\ alloy?}}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{SELECT\ ?equipment\ ?name\ ?power}
\codeline{WHERE\ \{}
\codeline{\hspace*{4\ttcw}?equipment\ a\ :Equipment\ ;}
\codeline{\hspace*{15\ttcw}:canProcess\ :Aluminum\ ;}
\codeline{\hspace*{15\ttcw}rdfs:label\ ?name\ .}
\codeline{\hspace*{4\ttcw}OPTIONAL\ \{\ ?equipment\ :power\ ?power\ \}}
\codeline{\hspace*{4\ttcw}FILTER(lang(?name)\ =\ "zh")}
\codeline{\}}
\codeline{ORDER\ BY\ ASC(?name)}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{\textcolor{cmtgray}{\#\ CQ2:\ 查询某工单的所有工序及所需设备}}
\codeline{\textcolor{cmtgray}{\#\ Find\ all\ operations\ and\ required\ equipment\ for\ a\ work\ order}}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{SELECT\ ?operation\ ?opLabel\ ?equipment\ ?eqLabel}
\codeline{WHERE\ \{}
\codeline{\hspace*{4\ttcw}:WorkOrder\_001\ :hasProcess\ ?process\ .}
\codeline{\hspace*{4\ttcw}?process\ :hasOperation\ ?operation\ .}
\codeline{\hspace*{4\ttcw}?operation\ rdfs:label\ ?opLabel\ ;}
\codeline{\hspace*{15\ttcw}:requires\ ?equipment\ .}
\codeline{\hspace*{4\ttcw}?equipment\ rdfs:label\ ?eqLabel\ .}
\codeline{\hspace*{4\ttcw}FILTER(lang(?opLabel)\ =\ "zh")}
\codeline{\hspace*{4\ttcw}FILTER(lang(?eqLabel)\ =\ "zh")}
\codeline{\}}
\codeline{ORDER\ BY\ ?operation}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{\textcolor{cmtgray}{\#\ CQ3:\ 查找所有可用（Idle状态）的加工设备}}
\codeline{\textcolor{cmtgray}{\#\ Find\ all\ available\ (Idle)\ processing\ equipment}}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{SELECT\ ?equipment\ ?name\ ?location}
\codeline{WHERE\ \{}
\codeline{\hspace*{4\ttcw}?equipment\ a\ :ProcessingEquipment\ ;}
\codeline{\hspace*{15\ttcw}:hasStatus\ :Idle\ ;}
\codeline{\hspace*{15\ttcw}rdfs:label\ ?name\ ;}
\codeline{\hspace*{15\ttcw}:locatedIn\ ?workCell\ .}
\codeline{\hspace*{4\ttcw}?workCell\ rdfs:label\ ?location\ .}
\codeline{\hspace*{4\ttcw}FILTER(lang(?name)\ =\ "zh")}
\codeline{\}}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{\textcolor{cmtgray}{\#\ CQ4:\ 统计各工作单元的设备数量}}
\codeline{\textcolor{cmtgray}{\#\ Count\ equipment\ per\ work\ cell}}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{SELECT\ ?workCell\ ?cellLabel\ (COUNT(?equipment)\ AS\ ?equipmentCount)}
\codeline{WHERE\ \{}
\codeline{\hspace*{4\ttcw}?equipment\ a\ :Equipment\ ;}
\codeline{\hspace*{15\ttcw}:locatedIn\ ?workCell\ .}
\codeline{\hspace*{4\ttcw}?workCell\ rdfs:label\ ?cellLabel\ .}
\codeline{\hspace*{4\ttcw}FILTER(lang(?cellLabel)\ =\ "zh")}
\codeline{\}}
\codeline{GROUP\ BY\ ?workCell\ ?cellLabel}
\codeline{ORDER\ BY\ DESC(?equipmentCount)}
\codeblank{}

\codeline{\hspace*{15\ttcw}:power\ "25.0"\^{}\^{}xsd:decimal\ ;}

\codeline{\}}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{\textcolor{cmtgray}{\#\ CQ6:\ SPARQL\ UPDATE\ -\ 条件更新设备状态}}
\codeline{\textcolor{cmtgray}{\#\ UPDATE\ equipment\ status\ conditionally}}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{DELETE\ \{\ ?equipment\ :hasStatus\ :Idle\ \}}
\codeline{INSERT\ \{\ ?equipment\ :hasStatus\ :Running\ \}}
\codeline{WHERE\ \{}
\codeline{\hspace*{4\ttcw}?equipment\ a\ :Equipment\ ;}
\codeline{\hspace*{15\ttcw}:hasStatus\ :Idle\ ;}
\codeline{\hspace*{15\ttcw}:assignedTo\ ?task\ .}
\codeline{\hspace*{4\ttcw}?task\ :hasStartTime\ ?start\ .}
\codeline{\hspace*{4\ttcw}FILTER(?start\ <=\ NOW())}
\codeline{\}}
\codeblank{}

\codeline{\}}
