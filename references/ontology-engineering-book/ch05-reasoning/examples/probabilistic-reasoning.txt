# 概率本体推理示例
# Probabilistic Ontology Reasoning Examples
#
# 经典描述逻辑只能表达"非真即假"的确定性知识
# 概率本体在公理上附加概率度量，用于处理工程中的不确定性

# ==============================
# 1. 概率类包含公理
# ==============================
# 在 TBox 公理上标注条件概率（P-SROIQ / PR-OWL 风格）

# 高功率设备需要维护的概率为85%：
P(NeedsMaintenance | HighPowerEquipment) = 0.85
# 含义：100台高功率设备中，约85台需要定期维护

# 长时间运行的设备出现精度漂移的概率为60%：
P(PrecisionDrift | LongRunning) = 0.60

# ==============================
# 2. 贝叶斯网络条件概率表 (CPT)
# ==============================
# 设备故障诊断贝叶斯网络结构：
#
#   HighTemp（高温）     LongRunning（长时间运行）
#          \                /
#           v              v
#             Fault（故障）
#
# 条件概率表：
P(Fault | HighTemp ∧ LongRunning)   = 0.90
P(Fault | HighTemp ∧ ¬LongRunning)  = 0.30
P(Fault | ¬HighTemp ∧ LongRunning)  = 0.10
P(Fault | ¬HighTemp ∧ ¬LongRunning) = 0.01

# 先验概率：
P(HighTemp) = 0.20      # 车间环境中设备高温的先验概率
P(LongRunning) = 0.35   # 设备长时间运行的先验概率

# ==============================
# 3. 概率推理计算示例
# ==============================
# 预测方向：观测到 Lathe_003 高温且长时间运行，故障概率多大？
# 直接查条件概率表：P(Fault) = 0.90 → 触发预防性维护工单

# 诊断方向：已知某设备发生故障，它处于高温状态的概率多大？
# 第一步，用全概率公式计算故障的边缘概率：
P(Fault) = 0.90×0.20×0.35 + 0.30×0.20×0.65
         + 0.10×0.80×0.35 + 0.01×0.80×0.65
         = 0.063 + 0.039 + 0.028 + 0.0052
         = 0.1352

# 第二步，计算高温条件下的故障概率：
P(Fault | HighTemp) = 0.90×0.35 + 0.30×0.65 = 0.51

# 第三步，用贝叶斯公式反推：
P(HighTemp | Fault) = P(Fault | HighTemp) × P(HighTemp) / P(Fault)
                    = (0.51 × 0.20) / 0.1352
                    ≈ 0.754

# 结论：故障设备中约75%与高温相关 → 维修时优先排查冷却系统

# ==============================
# 4. 马尔可夫逻辑网络 (MLN) 权重规则
# ==============================
# MLN 为一阶逻辑规则附加权重：权重越大，约束越"硬"
# 权重语义：满足该规则的可能世界，其概率按 exp(权重) 倍提升

2.5  HighPowerEquipment(x) → NeedsCooling(x)      # 强约束
1.2  CNCMachine(x) ∧ Aged(x) → PrecisionDrift(x)  # 中等约束
0.4  Idle(x) → Available(x)                       # 弱约束（默认倾向）

# 对比：硬约束（等价于无穷大权重）直接用 OWL 公理表达
Running ⊓ Maintenance ⊑ ⊥   # 运行中的设备不可同时维护（确定性）

# ==============================
# 5. 工程应用与工具
# ==============================
# 适用场景：
#   设备故障诊断与预测性维护（PHM）
#   质量风险评估（工艺参数 → 缺陷概率）
#   多源传感器数据融合（不确定信息合并）
#
# 代表工具 / 语言：
#   PR-OWL   —— 基于 MEBN 的概率 OWL 扩展
#   BayesOWL —— 将 OWL 本体转换为贝叶斯网络
#   Pronto   —— Pellet 推理器的概率扩展（P-SROIQ）
#   ProbLog / Alchemy —— 概率逻辑编程 / MLN 实现
#
# 实践建议：保持"确定性骨架 + 概率标注"分离——
# 类层次与硬约束用 OWL DL 表达，不确定性知识在概率层叠加，
# 避免将全部知识概率化导致推理复杂度失控
