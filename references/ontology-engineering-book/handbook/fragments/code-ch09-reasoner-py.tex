\codeline{\hspace*{8\ttcw}初始化推理引擎，加载本体文件}

\codeline{\hspace*{8\ttcw}Args:}

\codeline{\hspace*{8\ttcw}"""}

\codeblank{}
\codeline{\hspace*{4\ttcw}def\ run\_reasoner(self):}
\codeline{\hspace*{8\ttcw}"""}
\codeline{\hspace*{8\ttcw}执行Pellet推理器}
\codeline{\hspace*{8\ttcw}推理后，本体中会新增推导出的知识（隐含三元组）}
\codeline{\hspace*{8\ttcw}"""}

\codeline{\hspace*{12\ttcw}sync\_reasoner\_pellet(}

\codeline{\hspace*{12\ttcw})}
\codeline{\hspace*{8\ttcw}print("推理完成")}
\codeblank{}
\codeline{\hspace*{4\ttcw}def\ check\_conflicts(self)\ ->\ List[Dict]:}
\codeline{\hspace*{8\ttcw}"""}
\codeline{\hspace*{8\ttcw}检测所有任务冲突（运行SWRL规则推理后查询冲突实例）}
\codeblank{}
\codeline{\hspace*{8\ttcw}Returns:}
\codeline{\hspace*{12\ttcw}冲突列表，每个元素包含\ type,\ task,\ description}
\codeline{\hspace*{8\ttcw}"""}
\codeline{\hspace*{8\ttcw}\textcolor{cmtgray}{\#\ 运行推理以激活SWRL规则}}
\codeline{\hspace*{8\ttcw}self.run\_reasoner()}
\codeblank{}
\codeline{\hspace*{8\ttcw}conflicts\ =\ []}
\codeblank{}
\codeline{\hspace*{8\ttcw}\textcolor{cmtgray}{\#\ 查询能力冲突（设备无法加工所需材料）}}
\codeline{\hspace*{8\ttcw}if\ hasattr(self.onto,\ 'CapabilityConflict'):}
\codeline{\hspace*{12\ttcw}for\ c\ in\ list(self.onto.search(type=self.onto.CapabilityConflict)):}
\codeline{\hspace*{16\ttcw}conflicts.append(\{}
\codeline{\hspace*{20\ttcw}"type":\ "capability",}
\codeline{\hspace*{20\ttcw}"type\_zh":\ "能力冲突",}
\codeline{\hspace*{20\ttcw}"task":\ c.name,}
\codeline{\hspace*{20\ttcw}"description":\ f"任务\ \{c.name\}:\ 分配的设备无法加工所需材料"}
\codeline{\hspace*{16\ttcw}\})}
\codeblank{}
\codeline{\hspace*{8\ttcw}\textcolor{cmtgray}{\#\ 查询时间冲突（设备时间段重叠）}}
\codeline{\hspace*{8\ttcw}if\ hasattr(self.onto,\ 'TimeConflict'):}
\codeline{\hspace*{12\ttcw}for\ c\ in\ list(self.onto.search(type=self.onto.TimeConflict)):}
\codeline{\hspace*{16\ttcw}conflicts.append(\{}
\codeline{\hspace*{20\ttcw}"type":\ "time",}
\codeline{\hspace*{20\ttcw}"type\_zh":\ "时间冲突",}
\codeline{\hspace*{20\ttcw}"task":\ c.name,}
\codeline{\hspace*{20\ttcw}"description":\ f"任务\ \{c.name\}:\ 设备时间段存在重叠"}
\codeline{\hspace*{16\ttcw}\})}
\codeblank{}
\codeline{\hspace*{8\ttcw}\textcolor{cmtgray}{\#\ 查询状态冲突（设备处于Running状态）}}
\codeline{\hspace*{8\ttcw}if\ hasattr(self.onto,\ 'StatusConflict'):}
\codeline{\hspace*{12\ttcw}for\ c\ in\ list(self.onto.search(type=self.onto.StatusConflict)):}
\codeline{\hspace*{16\ttcw}conflicts.append(\{}
\codeline{\hspace*{20\ttcw}"type":\ "status",}
\codeline{\hspace*{20\ttcw}"type\_zh":\ "状态冲突",}
\codeline{\hspace*{20\ttcw}"task":\ c.name,}
\codeline{\hspace*{20\ttcw}"description":\ f"任务\ \{c.name\}:\ 设备当前状态不可用"}
\codeline{\hspace*{16\ttcw}\})}
\codeblank{}
\codeline{\hspace*{8\ttcw}return\ conflicts}
\codeblank{}
\codeline{\hspace*{4\ttcw}def\ recommend\_process(self,\ product\_name:\ str)\ ->\ List[Dict]:}
\codeline{\hspace*{8\ttcw}"""}
\codeline{\hspace*{8\ttcw}基于产品特征相似性推荐工艺}
\codeblank{}
\codeline{\hspace*{8\ttcw}Args:}
\codeline{\hspace*{12\ttcw}product\_name:\ 产品实例名称}
\codeblank{}
\codeline{\hspace*{8\ttcw}Returns:}
\codeline{\hspace*{12\ttcw}推荐工艺列表，按置信度排序}
\codeline{\hspace*{8\ttcw}"""}
\codeline{\hspace*{8\ttcw}product\ =\ self.onto.search\_one(iri=f"*\#\{product\_name\}")}
\codeline{\hspace*{8\ttcw}if\ not\ product:}
\codeline{\hspace*{12\ttcw}print(f"产品未找到:\ \{product\_name\}")}
\codeline{\hspace*{12\ttcw}return\ []}
\codeblank{}
\codeline{\hspace*{8\ttcw}self.run\_reasoner()}
\codeblank{}
\codeline{\hspace*{8\ttcw}recommendations\ =\ []}
\codeline{\hspace*{8\ttcw}if\ hasattr(product,\ 'recommendedProcess'):}
\codeline{\hspace*{12\ttcw}for\ proc\ in\ product.recommendedProcess:}
\codeline{\hspace*{16\ttcw}confidence\_vals\ =\ getattr(proc,\ 'confidence',\ [])}
\codeline{\hspace*{16\ttcw}confidence\ =\ confidence\_vals[0]\ if\ confidence\_vals\ else\ 0.8}
\codeline{\hspace*{16\ttcw}comment\_vals\ =\ getattr(proc,\ 'comment',\ [])}
\codeline{\hspace*{16\ttcw}description\ =\ str(comment\_vals[0])\ if\ comment\_vals\ else\ ''}
\codeline{\hspace*{16\ttcw}recommendations.append(\{}
\codeline{\hspace*{20\ttcw}"name":\ proc.name,}
\codeline{\hspace*{20\ttcw}"confidence":\ confidence,}
\codeline{\hspace*{20\ttcw}"description":\ description}
\codeline{\hspace*{16\ttcw}\})}
\codeblank{}
\codeline{\hspace*{8\ttcw}\textcolor{cmtgray}{\#\ 按置信度降序排序}}
\codeline{\hspace*{8\ttcw}recommendations.sort(key=lambda\ x:\ x['confidence'],\ reverse=True)}
\codeline{\hspace*{8\ttcw}return\ recommendations}
\codeblank{}
\codeline{\hspace*{4\ttcw}def\ get\_equipment\_for\_material(self,\ material\_name:\ str)\ ->\ List[str]:}
\codeline{\hspace*{8\ttcw}"""}
\codeline{\hspace*{8\ttcw}查询能加工指定材料的设备列表}
\codeblank{}
\codeline{\hspace*{8\ttcw}Args:}
\codeline{\hspace*{12\ttcw}material\_name:\ 材料类名称（如\ 'Aluminum'）}
\codeline{\hspace*{8\ttcw}"""}
\codeline{\hspace*{8\ttcw}material\_cls\ =\ getattr(self.onto,\ material\_name,\ None)}
\codeline{\hspace*{8\ttcw}if\ not\ material\_cls:}
\codeline{\hspace*{12\ttcw}print(f"材料类未找到:\ \{material\_name\}")}
\codeline{\hspace*{12\ttcw}return\ []}
\codeblank{}
\codeline{\hspace*{8\ttcw}results\ =\ []}
\codeline{\hspace*{8\ttcw}if\ hasattr(self.onto,\ 'Equipment'):}
\codeline{\hspace*{12\ttcw}for\ equipment\ in\ self.onto.Equipment.instances():}
\codeline{\hspace*{16\ttcw}for\ material\ in\ getattr(equipment,\ 'canProcess',\ []):}
\codeline{\hspace*{20\ttcw}if\ isinstance(material,\ material\_cls):}
\codeline{\hspace*{24\ttcw}\textcolor{cmtgray}{\#\ equipment.label\ is\ a\ LocalizedStringList;\ use\ index\ access}}
\codeline{\hspace*{24\ttcw}label\_list\ =\ equipment.label}
\codeline{\hspace*{24\ttcw}label\ =\ label\_list[0]\ if\ label\_list\ else\ equipment.name}
\codeline{\hspace*{24\ttcw}results.append(str(label))}
\codeline{\hspace*{24\ttcw}break}
\codeline{\hspace*{8\ttcw}return\ results}
