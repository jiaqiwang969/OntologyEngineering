\codeline{\textcolor{cmtgray}{\#\ SWRL调度规则（在Protégé\ SWRL\ Tab中输入）}}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ 规则1：能力冲突检测}}
\codeline{\textcolor{cmtgray}{\#\ Task(?t)\ \(\land\)\ requiresMaterial(?t,\ ?m)\ \(\land\)\ assignedTo(?t,\ ?e)\ \(\land\)}}
\codeline{\textcolor{cmtgray}{\#\ Equipment(?e)\ \(\land\)\ \(\lnot\)canProcess(?e,\ ?m)}}
\codeline{\textcolor{cmtgray}{\#\ \(\rightarrow\)\ CapabilityConflict(?t)}}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ 规则2：状态冲突检测}}
\codeline{\textcolor{cmtgray}{\#\ Equipment(?e)\ \(\land\)\ hasStatus(?e,\ Running)\ \(\land\)\ Task(?t)\ \(\land\)\ assignedTo(?t,\ ?e)}}
\codeline{\textcolor{cmtgray}{\#\ \(\rightarrow\)\ StatusConflict(?t)}}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ 规则3：时间冲突检测}}
\codeline{\textcolor{cmtgray}{\#\ Task(?t1)\ \(\land\)\ Task(?t2)\ \(\land\)\ assignedTo(?t1,\ ?e)\ \(\land\)\ assignedTo(?t2,\ ?e)\ \(\land\)}}
\codeline{\textcolor{cmtgray}{\#\ hasStartTime(?t1,\ ?s1)\ \(\land\)\ hasEndTime(?t1,\ ?e1)\ \(\land\)}}
\codeline{\textcolor{cmtgray}{\#\ hasStartTime(?t2,\ ?s2)\ \(\land\)\ hasEndTime(?t2,\ ?e2)\ \(\land\)}}
\codeline{\textcolor{cmtgray}{\#\ swrlb:lessThan(?s1,\ ?e2)\ \(\land\)\ swrlb:greaterThan(?e1,\ ?s2)}}
\codeline{\textcolor{cmtgray}{\#\ \(\rightarrow\)\ TimeConflict(?t1)}}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ 规则4：工艺推荐}}
\codeline{\textcolor{cmtgray}{\#\ Product(?p1)\ \(\land\)\ Product(?p2)\ \(\land\)}}
\codeline{\textcolor{cmtgray}{\#\ hasMaterial(?p1,\ ?m)\ \(\land\)\ hasMaterial(?p2,\ ?m)\ \(\land\)}}
\codeline{\textcolor{cmtgray}{\#\ hasFeature(?p1,\ ?f)\ \(\land\)\ hasFeature(?p2,\ ?f)\ \(\land\)}}
\codeline{\textcolor{cmtgray}{\#\ hasProcess(?p2,\ ?proc)}}
\codeline{\textcolor{cmtgray}{\#\ \(\rightarrow\)\ recommendedProcess(?p1,\ ?proc)}}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ 规则5：工序顺序}}
\codeline{\textcolor{cmtgray}{\#\ Operation(?op1)\ \(\land\)\ Operation(?op2)\ \(\land\)}}
\codeline{\textcolor{cmtgray}{\#\ precedes(?op1,\ ?op2)\ \(\land\)\ hasStatus(?op1,\ Completed)}}
\codeline{\textcolor{cmtgray}{\#\ \(\rightarrow\)\ canStart(?op2,\ true)}}
