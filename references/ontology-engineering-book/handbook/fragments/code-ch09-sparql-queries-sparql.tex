\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{\textcolor{cmtgray}{\#\ CQ1:\ 哪些设备可以加工铝合金？}}
\codeline{\textcolor{cmtgray}{\#\ Which\ equipment\ can\ process\ aluminum\ alloy?}}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{SELECT\ ?equipment\ ?name\ ?power\ ?location}
\codeline{WHERE\ \{}
\codeline{\hspace*{4\ttcw}?equipment\ a\ :Equipment\ ;}
\codeline{\hspace*{15\ttcw}:canProcess\ :Aluminum\ ;}
\codeline{\hspace*{15\ttcw}rdfs:label\ ?name\ .}
\codeline{\hspace*{4\ttcw}OPTIONAL\ \{\ ?equipment\ :power\ ?power\ \}}
\codeline{\hspace*{4\ttcw}OPTIONAL\ \{}
\codeline{\hspace*{8\ttcw}?equipment\ :locatedIn\ ?cell\ .}
\codeline{\hspace*{8\ttcw}?cell\ rdfs:label\ ?location\ .}
\codeline{\hspace*{4\ttcw}\}}
\codeline{\hspace*{4\ttcw}FILTER(lang(?name)\ =\ "zh")}
\codeline{\}}
\codeline{ORDER\ BY\ ASC(?name)}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{\textcolor{cmtgray}{\#\ CQ2:\ 当前哪些设备空闲可以接受任务？}}
\codeline{\textcolor{cmtgray}{\#\ Which\ equipment\ is\ currently\ idle?}}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{SELECT\ ?equipment\ ?name\ ?capability}
\codeline{WHERE\ \{}
\codeline{\hspace*{4\ttcw}?equipment\ a\ :ProcessingEquipment\ ;}
\codeline{\hspace*{15\ttcw}:hasStatus\ :Idle\ ;}
\codeline{\hspace*{15\ttcw}rdfs:label\ ?name\ .}
\codeline{\hspace*{4\ttcw}OPTIONAL\ \{}
\codeline{\hspace*{8\ttcw}?equipment\ :hasCapability\ ?cap\ .}
\codeline{\hspace*{8\ttcw}?cap\ rdfs:label\ ?capability\ .}
\codeline{\hspace*{4\ttcw}\}}
\codeline{\hspace*{4\ttcw}FILTER(lang(?name)\ =\ "zh")}
\codeline{\}}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{\textcolor{cmtgray}{\#\ CQ3:\ 查询所有冲突任务（SWRL推理后）}}
\codeline{\textcolor{cmtgray}{\#\ Find\ all\ conflicting\ tasks\ after\ inference}}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{SELECT\ ?task\ ?conflictType}
\codeline{WHERE\ \{}
\codeline{\hspace*{4\ttcw}\{}
\codeline{\hspace*{8\ttcw}?task\ a\ :CapabilityConflict\ .}
\codeline{\hspace*{8\ttcw}BIND("能力冲突：设备无法加工所需材料"\ AS\ ?conflictType)}
\codeline{\hspace*{4\ttcw}\}}
\codeline{\hspace*{4\ttcw}UNION}
\codeline{\hspace*{4\ttcw}\{}
\codeline{\hspace*{8\ttcw}?task\ a\ :TimeConflict\ .}
\codeline{\hspace*{8\ttcw}BIND("时间冲突：设备时间段重叠"\ AS\ ?conflictType)}
\codeline{\hspace*{4\ttcw}\}}
\codeline{\hspace*{4\ttcw}UNION}
\codeline{\hspace*{4\ttcw}\{}
\codeline{\hspace*{8\ttcw}?task\ a\ :StatusConflict\ .}
\codeline{\hspace*{8\ttcw}BIND("状态冲突：设备状态不可用"\ AS\ ?conflictType)}
\codeline{\hspace*{4\ttcw}\}}
\codeline{\hspace*{4\ttcw}UNION}
\codeline{\hspace*{4\ttcw}\{}
\codeline{\hspace*{8\ttcw}?task\ a\ :ResourceConflict\ .}
\codeline{\hspace*{8\ttcw}BIND("资源冲突：资源可用性不足"\ AS\ ?conflictType)}
\codeline{\hspace*{4\ttcw}\}}
\codeline{\}}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{\textcolor{cmtgray}{\#\ CQ4:\ 给定产品，查询推荐工艺路线}}
\codeline{\textcolor{cmtgray}{\#\ Get\ recommended\ process\ for\ a\ product}}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{SELECT\ ?product\ ?process\ ?processLabel\ ?confidence}
\codeline{WHERE\ \{}
\codeline{\hspace*{4\ttcw}?product\ a\ :Product\ ;}
\codeline{\hspace*{13\ttcw}:recommendedProcess\ ?process\ .}
\codeline{\hspace*{4\ttcw}?process\ rdfs:label\ ?processLabel\ .}
\codeline{\hspace*{4\ttcw}OPTIONAL\ \{\ ?process\ :confidence\ ?confidence\ \}}
\codeline{\hspace*{4\ttcw}FILTER(?product\ =\ :Product\_001)}
\codeline{\hspace*{4\ttcw}FILTER(lang(?processLabel)\ =\ "zh")}
\codeline{\}}
\codeline{ORDER\ BY\ DESC(?confidence)}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{\textcolor{cmtgray}{\#\ CQ5:\ 工单的工序执行顺序}}
\codeline{\textcolor{cmtgray}{\#\ Get\ correct\ operation\ sequence\ for\ a\ work\ order}}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{SELECT\ ?operation\ ?opLabel\ ?startTime\ ?status}
\codeline{WHERE\ \{}
\codeline{\hspace*{4\ttcw}:WorkOrder\_001\ :hasProcess\ ?process\ .}
\codeline{\hspace*{4\ttcw}?process\ :hasOperation\ ?operation\ .}
\codeline{\hspace*{4\ttcw}?operation\ rdfs:label\ ?opLabel\ ;}
\codeline{\hspace*{15\ttcw}:hasStatus\ ?status\ .}
\codeline{\hspace*{4\ttcw}OPTIONAL\ \{\ ?operation\ :hasStartTime\ ?startTime\ \}}
\codeline{\hspace*{4\ttcw}FILTER(lang(?opLabel)\ =\ "zh")}
\codeline{\}}
\codeline{ORDER\ BY\ ?startTime}
\codeblank{}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{\textcolor{cmtgray}{\#\ CQ6:\ 设备利用率统计}}
\codeline{\textcolor{cmtgray}{\#\ Equipment\ utilization\ statistics}}
\codeline{\textcolor{cmtgray}{\#\ ==============================}}
\codeline{SELECT\ ?equipment\ ?name\ (COUNT(?task)\ AS\ ?taskCount)}
\codeline{WHERE\ \{}
\codeline{\hspace*{4\ttcw}?equipment\ a\ :Equipment\ ;}
\codeline{\hspace*{15\ttcw}rdfs:label\ ?name\ .}
\codeline{\hspace*{4\ttcw}OPTIONAL\ \{\ ?task\ :assignedTo\ ?equipment\ \}}
\codeline{\hspace*{4\ttcw}FILTER(lang(?name)\ =\ "zh")}
\codeline{\}}
\codeline{GROUP\ BY\ ?equipment\ ?name}
\codeline{ORDER\ BY\ DESC(?taskCount)}
\codeblank{}
